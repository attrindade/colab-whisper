def verificar_etapas(passo_atual):
    etapas = {
        1: [],
        '2a': [1],
        '2b': [1],
        '2c': [1],
        '3.1': [1, '2a', '2b', '2c'],
        '3.2': [1, '2a', '2b', '2c', '3.1'],
        4: [1, '2a', '2b', '2c', '3.1']
    }

    if passo_atual != 1:
        etapas_concluidas = globals()['etapas_concluidas']

        if passo_atual not in etapas:
            raise ValueError(f"Passo {passo_atual} não reconhecido")

        # Se estamos iniciando um novo passo 2, resetamos para uma nova transcrição
        if passo_atual in ['2a', '2b', '2c'] and any(etapa in etapas_concluidas for etapa in ['2a', '2b', '2c', '3.1', '3.2', '4']):
            print("Detectada iniciação de nova transcrição.")
            reset_para_nova_transcricao()
            etapas_concluidas = globals()['etapas_concluidas']

        for pre_requisito in etapas[passo_atual]:
            if isinstance(pre_requisito, str) and pre_requisito[0] == '2':
                if not any(etapa in etapas_concluidas for etapa in ['2a', '2b', '2c']):
                    print(f"Erro: Você deve completar um dos passos 2 (a, b ou c) antes do passo {passo_atual}")
                    sys.exit(1)
            elif pre_requisito not in etapas_concluidas:
                print(f"Erro: Você deve completar o passo {pre_requisito} antes do passo {passo_atual}")
                sys.exit(1)

    print(f"Verificação para o passo {passo_atual} concluída com sucesso.")

def reset_para_nova_transcricao():
    etapas_concluidas = globals().get('etapas_concluidas', set())
    etapas_para_remover = {'2a', '2b', '2c', '3.1', '3.2', '4'}
    etapas_concluidas = etapas_concluidas - etapas_para_remover
    globals()['etapas_concluidas'] = etapas_concluidas

    # Resetar outras variáveis globais relevantes
    if 'result' in globals():
        del globals()['result']
    if 'ssm' in globals():
        del globals()['ssm']
    if 'ssm_mod' in globals():
        del globals()['ssm_mod']

    print("Sistema resetado para uma nova transcrição.")

def get_video_info(link):
  # Obtem o título do vídeo
  title_process = subprocess.run(["yt-dlp", "--get-title", "--no-warnings", link], capture_output=True, text=True)

def converter_para_segundos(tempo_str):
  horas, minutos, segundos = map(int, tempo_str.split(':'))
  return horas * 3600 + minutos * 60 + segundos

def validar_tempo(tempo_inicio, tempo_fim):
  formato_correto = "%H:%M:%S"

  try:
    datetime.strptime(tempo_inicio, formato_correto)
  except ValueError:
    logger.error("O tempo de início não está no formato HH:MM:SS")
    logger.debug(f'Início: "{tempo_inicio}", Fim: "{tempo_fim}"')
    return 0, "Erro: O tempo de início deve estar no formato HH:MM:SS.\n"

  if tempo_fim in ["", "00:00:00"]:
    logger.info("Tempo validado com sucesso.")
    logger.debug(f'Início: "{tempo_inicio}", Fim: "{tempo_fim}"')
    return 2, "Validação bem-sucedida: o vídeo deve ir até o fim."

  try:
    datetime.strptime(tempo_fim, formato_correto)
  except ValueError:
    logger.error("O tempo de fim não está no formato HH:MM:SS")
    logger.debug(f'Início: "{tempo_inicio}", Fim: "{tempo_fim}"')
    return 0, "Erro: O tempo de fim deve estar no formato HH:MM:SS.\n"

  inicio_segundos = converter_para_segundos(tempo_inicio)
  fim_segundos = converter_para_segundos(tempo_fim)

  if inicio_segundos < fim_segundos:
      logger.info("Tempo validado com sucesso. Início menor que fim.")
      logger.debug(f'Início: "{tempo_inicio}", Fim: "{tempo_fim}"')
      return 1, "Validação bem-sucedida: o tempo de início é menor que o tempo de fim."
  else:
      logger.error("Erro na validação. Início não é menor que fim.")
      logger.debug(f'Início: "{tempo_inicio}", Fim: "{tempo_fim}"')
      return 0, "Erro: O tempo de início deve ser menor que o tempo de fim.\n"

#### Função para modificar o embed do onedrive, transformá-lo em link direto e fazer o download

def process_embed_url(embed_url):
  # Coloque aqui a lógica de processamento usando `embed_url`
  url_match = re.search(r'src="([^"]+)"', embed_url)
  if url_match:
      embed_url = url_match.group(1)
      download_url = embed_url.replace("embed", "download")
      vocal_target_filename = baixar_arquivo(download_url)
      vocal_target = move_para_pasta_input(vocal_target_filename)
      return vocal_target, vocal_target_filename
  else:
      display(Markdown('## Erro: ❌ Verifique se o texto inserido é o texto do "Embed" do arquivo no OneDrive.'))
      display(Markdown('## Erro: ❌ Veja como pegar o "Embed" no vídeo disponível na instrução desse passo.'))
      raise

  if cd:
      # Cobrindo casos com 'filename*' (codificação extendida RFC 5987)
      fname_ext = re.findall('filename\*=.*\'\'(.+)', cd)
      if fname_ext:
          nome_arquivo = unquote(fname_ext[0])
      else:
          # Tentativa para casos com 'filename='
          fname = re.findall('filename="?([^";]+)"?', cd)
          if fname:
              nome_arquivo = fname[0]

  # Verifica se a resposta é bem-sucedida (código de status HTTP 200)
  if resposta.status_code == 200:
      # Abre um novo arquivo em modo de escrita binária no caminho de destino e grava o conteúdo
      with open(nome_arquivo, 'wb') as arquivo:
          for chunk in resposta.iter_content(chunk_size=128):
              arquivo.write(chunk)
      print(f"Arquivo '{nome_arquivo}' baixado com sucesso.")
      return nome_arquivo
  else:
      print("Erro ao baixar o arquivo.")

def get_word_ts_anchor(s, e, option="start"):
  if option == "end":
      return e
  elif option == "mid":
      return (s + e) / 2
  return s

def get_words_speaker_mapping(wrd_ts, spk_ts, word_anchor_option="start"):
  s, e, sp = spk_ts[0]
  wrd_pos, turn_idx = 0, 0
  wrd_spk_mapping = []
  for wrd_dict in wrd_ts:
      ws, we, wrd = (
          int(wrd_dict["start"] * 1000),
          int(wrd_dict["end"] * 1000),
          wrd_dict["word"],
      )
      wrd_pos = get_word_ts_anchor(ws, we, word_anchor_option)
      while wrd_pos > float(e):
          turn_idx += 1
          turn_idx = min(turn_idx, len(spk_ts) - 1)
          s, e, sp = spk_ts[turn_idx]
          if turn_idx == len(spk_ts) - 1:
              e = get_word_ts_anchor(ws, we, option="end")
      wrd_spk_mapping.append(
          {"word": wrd, "start_time": ws, "end_time": we, "speaker": sp}
      )
  return wrd_spk_mapping

sentence_ending_punctuations = ".?!"

def get_first_word_idx_of_sentence(word_idx, word_list, speaker_list, max_words):
  is_word_sentence_end = (
      lambda x: x >= 0 and word_list[x][-1] in sentence_ending_punctuations
  )
  left_idx = word_idx
  while (
      left_idx > 0
      and word_idx - left_idx < max_words
      and speaker_list[left_idx - 1] == speaker_list[left_idx]
      and not is_word_sentence_end(left_idx - 1)
  ):
      left_idx -= 1

  return left_idx if left_idx == 0 or is_word_sentence_end(left_idx - 1) else -1

def get_last_word_idx_of_sentence(word_idx, word_list, max_words):
  is_word_sentence_end = (
      lambda x: x >= 0 and word_list[x][-1] in sentence_ending_punctuations
  )
  right_idx = word_idx
  while (
      right_idx < len(word_list)
      and right_idx - word_idx < max_words
      and not is_word_sentence_end(right_idx)
  ):
      right_idx += 1

  return (
      right_idx
      if right_idx == len(word_list) - 1 or is_word_sentence_end(right_idx)
      else -1
  )

def get_realigned_ws_mapping_with_punctuation(word_speaker_mapping, max_words_in_sentence=50):
  is_word_sentence_end = (
      lambda x: x >= 0
      and word_speaker_mapping[x]["word"][-1] in sentence_ending_punctuations
  )
  wsp_len = len(word_speaker_mapping)

  words_list, speaker_list = [], []
  for k, line_dict in enumerate(word_speaker_mapping):
      word, speaker = line_dict["word"], line_dict["speaker"]
      words_list.append(word)
      speaker_list.append(speaker)

  k = 0
  while k < len(word_speaker_mapping):
      line_dict = word_speaker_mapping[k]
      if (
          k < wsp_len - 1
          and speaker_list[k] != speaker_list[k + 1]
          and not is_word_sentence_end(k)
      ):
          left_idx = get_first_word_idx_of_sentence(
              k, words_list, speaker_list, max_words_in_sentence
          )
          right_idx = (
              get_last_word_idx_of_sentence(
                  k, words_list, max_words_in_sentence - k + left_idx - 1
              )
              if left_idx > -1
              else -1
          )
          if min(left_idx, right_idx) == -1:
              k += 1
              continue

          spk_labels = speaker_list[left_idx : right_idx + 1]
          mod_speaker = max(set(spk_labels), key=spk_labels.count)
          if spk_labels.count(mod_speaker) < len(spk_labels) // 2:
              k += 1
              continue

          speaker_list[left_idx : right_idx + 1] = [mod_speaker] * (
              right_idx - left_idx + 1
          )
          k = right_idx

      k += 1

  k, realigned_list = 0, []
  while k < len(word_speaker_mapping):
      line_dict = word_speaker_mapping[k].copy()
      line_dict["speaker"] = speaker_list[k]
      realigned_list.append(line_dict)
      k += 1

  return realigned_list

def get_sentences_speaker_mapping(word_speaker_mapping, spk_ts):
  sentence_checker = nltk.tokenize.PunktSentenceTokenizer().text_contains_sentbreak
  s, e, spk = spk_ts[0]
  prev_spk = spk

  snts = []
  snt = {"speaker": f"Locutor {spk}", "start_time": s, "end_time": e, "text": ""}

  for wrd_dict in word_speaker_mapping:
      wrd, spk = wrd_dict["word"], wrd_dict["speaker"]
      s, e = wrd_dict["start_time"], wrd_dict["end_time"]
      if spk != prev_spk or sentence_checker(snt["text"] + " " + wrd):
          snts.append(snt)
          snt = {
              "speaker": f"Locutor {spk}",
              "start_time": s,
              "end_time": e,
              "text": "",
          }
      else:
          snt["end_time"] = e
      snt["text"] += wrd + " "
      prev_spk = spk

  snts.append(snt)
  return snts

def format_timestamp(seconds: float, always_include_hours: bool = False, decimal_marker: str = ".", exclude_miliseconds: bool = False):
  assert seconds >= 0, "non-negative timestamp expected"

  hours = seconds // 3600
  seconds -= hours * 3600

  minutes = seconds // 60
  seconds -= minutes * 60

  # Se você ainda quiser lidar com milissegundos (parte decimal dos segundos)
  milliseconds = int((seconds % 1) * 1000)
  seconds = int(seconds)  # Pegar apenas a parte inteira dos segundos

  hours_marker = f"{int(hours):02d}:" if always_include_hours or hours > 0 else ""

  if exclude_miliseconds:
      return f"{hours_marker}{int(minutes):02d}:{int(seconds):02d}"
  else:
      return f"{hours_marker}{int(minutes):02d}:{int(seconds):02d}{decimal_marker}{milliseconds:03d}"

def update_speaker_names(sentences, speaker_names):
    """
    Atualiza os nomes dos locutores nas sentenças com os nomes fornecidos pelo usuário.

    :param sentences: Lista de dicionários, onde cada dicionário representa uma sentença e contém informações sobre o locutor.
    :param speaker_names: Dicionário com o mapeamento dos locutores antigos para os novos nomes fornecidos pelo usuário.
    :return: Lista de sentenças com os nomes dos locutores atualizados.
    """
    for sentence in sentences:
        # Verifica se a chave 'speaker' existe e se não existe, atribui 'Unknown Speaker'
        current_speaker = sentence.get("speaker", "Unknown Speaker")

        # Verifica se o locutor atual tem um novo nome fornecido pelo usuário
        if current_speaker in speaker_names:
            # Atualiza o nome do locutor na sentença
            sentence["speaker"] = speaker_names[current_speaker]

    return sentences

def write_txt(sentences_speaker_mapping, f, locutores: bool = False, timestamp_linha: bool = False):
  if locutores:
    previous_speaker = sentences_speaker_mapping[0]["speaker"]
    start_time = sentences_speaker_mapping[0]["start"]
    f.write(f"{previous_speaker} {format_timestamp(start_time, exclude_miliseconds = True)}:\n")

  for sentence_dict in sentences_speaker_mapping:
      sentence = sentence_dict["text"].strip()
      start_time = sentence_dict['start']
      if locutores:
        speaker = sentence_dict["speaker"]
        # If this speaker doesn't match the previous one, start a new paragraph
        if speaker != previous_speaker:
          f.write(f"\n{speaker} {format_timestamp(start_time, exclude_miliseconds = True)}:\n")
          previous_speaker = speaker
      # No matter what, write the current sentence
      if timestamp_linha:
        f.write(f"[{format_timestamp(start_time, exclude_miliseconds = True)}] {sentence}\n")
      else:
        f.write(sentence + "\n")

def write_srt(transcript, file1, locutores: bool = False):
  """
  Write a transcript to a file in SRT format.

  """
  for i, segment in enumerate(transcript, start=1):
      start_time = format_timestamp(segment['start'], always_include_hours=True, decimal_marker=',')
      end_time = format_timestamp(segment['end'], always_include_hours=True, decimal_marker=',')
      text = segment['text'].strip().replace('-->', '->')

      if locutores:
          line = f"{segment['speaker']}: {text}"
      else:
          line = text

      srt_block = f"{i}\n{start_time} --> {end_time}\n{line}\n"
      print(srt_block, file=file1, flush=True)

def write_nvivo_tabbed(transcript, file1):
    """
    Escreve a transcrição em formato NVivo com colunas separadas por TAB:
    Período <TAB> Conteúdo <TAB> Speaker
    Usa timestamps relativos (sem UTC+8), formato m:ss,d com uma casa decimal.
    """
    def format_m_ss_d(seconds):
        minutes = int(seconds // 60)
        remaining = seconds % 60
        return f"{minutes}:{remaining:04.1f}".replace('.', ',')

    # Cabeçalho
    print("Período\tConteúdo\tSpeaker", file=file1)

    for segment in transcript:
        start = format_m_ss_d(segment["start"])
        end = format_m_ss_d(segment["end"])
        text = segment["text"].strip().replace('-->', '')
        speaker = segment.get("speaker", "Unknown")
        print(f"{start} - {end}\t{text}\t{speaker}", file=file1)


def find_numeral_symbol_tokens(tokenizer):
  numeral_symbol_tokens = [
      -1,
  ]
  for token, token_id in tokenizer.get_vocab().items():
      has_numeral_symbol = any(c in "0123456789%$£" for c in token)
      if has_numeral_symbol:
          numeral_symbol_tokens.append(token_id)
  return numeral_symbol_tokens

def _get_next_start_timestamp(word_timestamps, current_word_index, final_timestamp):
  # if current word is the last word
  if current_word_index == len(word_timestamps) - 1:
      return word_timestamps[current_word_index]["start"]

  next_word_index = current_word_index + 1
  while current_word_index < len(word_timestamps) - 1:
      if word_timestamps[next_word_index].get("start") is None:
          # if next word doesn't have a start timestamp
          # merge it with the current word and delete it
          word_timestamps[current_word_index]["word"] += (
              " " + word_timestamps[next_word_index]["word"]
          )

          word_timestamps[next_word_index]["word"] = None
          next_word_index += 1
          if next_word_index == len(word_timestamps):
              return final_timestamp

      else:
          return word_timestamps[next_word_index]["start"]

def filter_missing_timestamps(word_timestamps, initial_timestamp=0, final_timestamp=None):
  # handle the first and last word
  if word_timestamps[0].get("start") is None:
      word_timestamps[0]["start"] = (
          initial_timestamp if initial_timestamp is not None else 0
      )
      word_timestamps[0]["end"] = _get_next_start_timestamp(
          word_timestamps, 0, final_timestamp
      )

  result = [
      word_timestamps[0],
  ]

  for i, ws in enumerate(word_timestamps[1:], start=1):
      # if ws doesn't have a start and end
      # use the previous end as start and next start as end
      if ws.get("start") is None and ws.get("word") is not None:
          ws["start"] = word_timestamps[i - 1]["end"]
          ws["end"] = _get_next_start_timestamp(word_timestamps, i, final_timestamp)

      if ws["word"] is not None:
          result.append(ws)
  return result

def cleanup(path: str):
  """path could either be relative or absolute."""
  # check if file or directory exists
  if os.path.isfile(path) or os.path.islink(path):
      # remove file
      os.remove(path)
  elif os.path.isdir(path):
      # remove directory and all its content
      shutil.rmtree(path)
  else:
      raise ValueError("Path {} is not a file or dir.".format(path))

def transcribe(
  audio_file: str,
  language: str,
  model_name: str,
  compute_dtype: str,
  suppress_numerals: bool,
  device: str,
):
  from faster_whisper import WhisperModel

  # Faster Whisper non-batched
  # Run on GPU with FP16
  whisper_model = WhisperModel(model_name, device=device, compute_type=compute_dtype)

  # or run on GPU with INT8
  # model = WhisperModel(model_size, device="cuda", compute_type="int8_float16")
  # or run on CPU with INT8
  # model = WhisperModel(model_size, device="cpu", compute_type="int8")

  if suppress_numerals:
      numeral_symbol_tokens = find_numeral_symbol_tokens(whisper_model.hf_tokenizer)
  else:
      numeral_symbol_tokens = None

  if language is not None and language in wav2vec2_langs:
      word_timestamps = False
  else:
      word_timestamps = True

  segments, info = whisper_model.transcribe(
      audio_file,
      language=language,
      beam_size=5,
      word_timestamps=word_timestamps,  # TODO: disable this if the language is supported by wav2vec2
      suppress_tokens=numeral_symbol_tokens,
      vad_filter=True,
  )
  whisper_results = []
  for segment in segments:
      whisper_results.append(segment._asdict())
  # clear gpu vram
  del whisper_model
  torch.cuda.empty_cache()
  return whisper_results, language


def transcribe_batched(
  audio_file: str,
  language: str,
  batch_size: int,
  model_name: str,
  compute_dtype: str,
  suppress_numerals: bool,
  device: str,
):
  import whisperx

  # Faster Whisper batched
  whisper_model = whisperx.load_model(
      model_name,
      device,
      compute_type=compute_dtype,
      asr_options={"suppress_numerals": suppress_numerals},
  )
  audio = whisperx.load_audio(audio_file)
  result = whisper_model.transcribe(audio, language=language, batch_size=batch_size, print_progress=True)
  del whisper_model
  torch.cuda.empty_cache()
  return result["segments"], result["language"]
